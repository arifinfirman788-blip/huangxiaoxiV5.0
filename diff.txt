+                          onClick={() => !isAdded && addWidget(widget.id)}
+                        >
+                          <div className="bg-gray-50 p-1.5 rounded-full mb-1">{widget.icon}</div>
+                          <div className="text-[10px] font-bold text-gray-800 text-center leading-tight">{widget.title}</div>
+                          {isAdded && <div className="absolute top-1 right-1 w-2 h-2 bg-indigo-500 rounded-full" />}
+                        </div>
+                      );
+                    })}
+                  </div>
+                </div>

     </div>
